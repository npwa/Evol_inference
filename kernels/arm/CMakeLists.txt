cmake_minimum_required(VERSION 3.20)
project(qdot CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Bit-exactness between the reference and the NEON kernels requires no fused multiply-add contraction.
add_compile_options(-O2 -ffp-contract=off -Wall -Wextra)
add_library(qdot STATIC qdot.cpp)
add_executable(qdot_test qdot_test.cpp)
target_link_libraries(qdot_test qdot)
add_executable(qdot_bench qdot_bench.cpp)
target_link_libraries(qdot_bench qdot)
enable_testing()
add_test(NAME qdot_test COMMAND qdot_test)
